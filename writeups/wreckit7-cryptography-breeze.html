Deskripsi Challenge

angin sepoi-sepoi

Identifikasi Kerentanan

Diberikan dua file, yaitu chall.py dan out.txt.

Pada Python, randomizer bawaan memakai algoritma MT19937 yang memiliki internal state sebanyak 624 integer 32-bit. Di baris 45, P0 menyimpan angka acak dengan memanggil A() selama 624 kali berturut-turut. P0 inilah yang nantinya dicetak pada out.txt sebagai o. Dari hasil tersebut kita bisa melakukan reverse engineering untuk setiap nilai.

Selanjutnya, fungsi pengacakan menghasilkan angka acak dengan mengurangi x dengan last, lalu menggabungkannya dengan prev. Meskipun pengacakan dilakukan berkali-kali, algoritma ini tetap linear sehingga state-nya bisa dibongkar untuk menebak angka selanjutnya.

Celah lain ada pada bit x di posisi paling belakang setelah dilakukan y << 1. Karena y pasti genap, angka ganjil-genap pada hasil akhir sc dapat diketahui dari nilai x.

Reasoning

Pendekatan yang digunakan relevan karena setiap algoritma yang dipecahkan akan berguna untuk pemecahan algoritma berikutnya, sehingga proses dekripsi bisa lebih efisien.

Langkah-langkah

Untuk mempermudah kloning MT19937, saya menggunakan library randcrack.

  1. Kloning state MT19937 dari 624 nilai o menggunakan RandCrack.
  2. Simulasikan ulang pembangkitan parameter challenge memakai fungsi B, C, dan D untuk mendapatkan nilai-nilai yang diperlukan.
  3. Lakukan discrete logarithm menggunakan sympy untuk memulihkan nilai high dan low.
  4. Bruteforce ml berukuran 16-bit bawah memakai 16 indeks awal sc.
  5. Bruteforce mr berukuran 16-bit atas menggunakan ml yang sudah ditemukan dan sisa data sc.
  6. Dekripsi AES dengan kandidat key yang dihasilkan.

Solver

import random, math
from randcrack import RandCrack
from sympy.ntheory import discrete_log
from hashlib import sha256
from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad
from collections import Counter


data = {}
exec(open("out.txt").read(), {}, data)
o, p, g, le, sc, ct = data['o'], data['n'], data['r'], data['le'], data['sc'], bytes.fromhex(data['ct'])


cracker = RandCrack()
for x in o:
    cracker.submit(x)


A = lambda: cracker.predict_getrandbits(32)


def B(n):
    x = 0
    while n:
        k = min(n, A().bit_length() or 1)
        if k > 32:
            k = 32
        x, n = (x << k) | (A() >> (32 - k)), n - k
    return x


def C(n):
    k = n.bit_length()
    while 1:
        x = B(k)
        if x < n:
            return x


def D(v):
    v = list(v)
    for i in range(len(v) - 1, 0, -1):
        j = C(i + 1)
        v[i], v[j] = v[j], v[i]
    return v


U = lambda: ((A() << 32) ^ A())


w0 = next(w for _ in iter(int, 1) if 48 <= (w := A().bit_length() + A().bit_length()) <= 80)
m0, z0 = 1 << w0, (1 << w0) - 1
X = lambda t: ((U() ^ (t * 0x9e3779b97f4a7c15)) & z0)


def G(t, o=0, u=1):
    x = X(t)
    y = (U() ^ ((x << 9) & z0) ^ (x >> 13) ^ (t << 17)) & z0
    p_val, q_val = 2 + (x & 31), 4 + ((y >> 7) & 31)
    v = [(p_val + i + ((x >> (i & 31)) & 1) + ((y >> ((i + 5) & 31)) & 1)) * u for i in range(q_val)]
    if o:
        v = [(r << 1) | 1 for r in v]
    return D(v)


H = lambda t, o=0, u=1: (v := G(t, o, u))[C(len(v))]

a0, a1 = H(A()), H(A())
while not 20 <= a0 + a1 <= 38:
    a0, a1 = H(A()), H(A())
a2, a3, a4 = a0 + a1, H(A()) + H(A()), H(A(), 1)

while 1:
    a5 = B(a3) | 1
    if a5 > 3 and pow(a5, a4, m0) not in (1, z0):
        break

a7 = next(a for _ in iter(int, 1) if 104 <= (a := H(A(), 0, 4)) <= 184)
a8, a9 = H(A()), H(A())
aa = [1 + C(H(A())) for _ in range(a7)]
ab, ac, ad, ae = C(4), C(4), C(4), C(3)
af, ag, ah, ai, aj, ak, al = B(a2), B(a2) | 1, B(a2), C(a2), C(a2), B(a2), B(a2)
am = next(a for _ in iter(int, 1) if 30 <= (a := H(A())) <= 40)
an, ao = p.bit_length(), C(p.bit_length())
ap, aq, ar = B(an), B(an) | 1, B(an)
H(A())


L_rot = lambda x, s, w: x if s == 0 else ((x << s) | (x >> (w - s))) & ((1 << w) - 1)
O_rot = lambda x, s, w: x if s == 0 else ((x >> s) | (x << (w - s))) & ((1 << w) - 1)


def invert_cut(val, idx):
    y = O_rot(val, ao, an)
    if ad & 2:
        y ^= al
    if ad & 1:
        y ^= ak
    if ac & 2:
        y = L_rot(y, aj, an)
    if ac & 1:
        y ^= (ag * idx + ah) & ((1 << an) - 1)
    if ab & 2:
        y = O_rot(y, ai, an)
    if ab & 1:
        y ^= af
    return y


total_calls = a8 + a9 + 2
k_indices = []
for j in aa:
    k_indices.append(total_calls + j)
    total_calls += j

A_base_am = pow(a5, a4, m0) % (1 << am)
A_inv_am = pow(A_base_am, -1, 1 << am)

H_cands = []
for idx, val in enumerate(le):
    Y = invert_cut(val, idx)
    if Y < p:
        try:
            E_k = discrete_log(p, Y, g)
            H_cands.append((E_k * pow(A_inv_am, k_indices[idx], 1 << am)) % (1 << am))
        except ValueError:
            pass

most_common_H = Counter(H_cands).most_common(1)[0][0]

lo0 = (most_common_H * pow(A_base_am, a8 + 1, 1 << am)) % (1 << am)
lo1 = (most_common_H * pow(A_base_am, a8 + a9 + 2, 1 << am)) % (1 << am)


HW = lambda x: bin(x).count('1')
ml = next(ml_cand for ml_cand in range(65536) if all(sc[i] - (HW((ml_cand * (i + 1) + lo0 + (lo1 << (i & 7))) & 0xffff) << 1) - ((ml_cand >> i) & 1) in (0, 1, 2) for i in range(16)))

y_vals = [HW((ml * (i + 1) + lo0 + (lo1 << (i & 7))) & 0xffff) for i in range(32)]
valid_mr = [mr_cand for mr_cand in range(65536) if all(sc[i] - (y_vals[i] << 1) - ((mr_cand >> (i - 16)) & 1) in (0, 1, 2) for i in range(16, 32))]


a_bytes = (lo0 ^ ml).to_bytes(8, "big")
for mr in valid_mr:
    for rand_byte in range(11):
        key = sha256(a_bytes + (lo1 ^ mr).to_bytes(8, "big") + bytes([rand_byte])).digest()[:16]
        try:
            flag = unpad(AES.new(key, AES.MODE_ECB).decrypt(ct), 16)
            if all(32 <= c < 127 for c in flag):
                print(flag.decode('utf-8'))
                exit()
        except:
            pass
